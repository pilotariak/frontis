Quick start

curl 
# List clubs for the LCAPB league
curl -s -X POST http://localhost:4000/graphql \
  -H "Content-Type: application/json" \
  -H "X-Pilotariak-League: lcapb" \
  -d '{"query":"{ clubs { id name } }"}' | jq

# Cross-subgraph: results + club names (federation in action)
curl -s -X POST http://localhost:4000/graphql \
  -H "Content-Type: application/json" \
  -H "X-Pilotariak-League: lcapb" \
  -d '{"query":"{ results { id scoreA scoreB clubA { name } clubB { name } } }"}' | jq
